# DriftAnalyser PCB v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Generate a complete KiCad 9 project (schematic, PCB, JLCPCB fab files) for the DriftAnalyser v2 board from a single Python design model, verified by KiCad's own ERC/DRC/parity checks.

**Architecture:** One Python "design model" (`PCB/v2/gen/model.py`) is the source of truth: every part with its symbol, footprint, LCSC number and pin→net map. Two writers consume it: `sch_writer.py` emits `.kicad_sch` files by hand (S-expressions, symbols copied from KiCad's bundled libraries plus a small custom lib), and `pcb_builder.py` builds the `.kicad_pcb` through KiCad's `pcbnew` Python API (footprints, nets, placement, zones, outline, silkscreen). Routing of signal nets is done by Freerouting through DSN/SES; power nets are zones/wide tracks written by the builder. `kicad-cli` does ERC, DRC with schematic parity, netlist export, gerbers, positions, BOM.

**Tech Stack:** KiCad 9 (brew cask, `kicad-cli`, bundled Python with `pcbnew`), Python 3 (system `python3` for pure-python code and tests, `pytest`), Freerouting jar + Temurin JDK 21 for autorouting.

**Spec:** `docs/superpowers/specs/2026-09-06-pcb-v2-design.md`

## Global Constraints

- Only 12 V strips. Strip connector pinout: pin1 `12V_LED`, pin2 `LEDn_DATA`, pin3 `GND`. Six connectors `J_LED1..J_LED6`, JST VH B3P-VH.
- Power input XT60PW-M. Nets `VBAT` (battery, protected by SMDJ18A) and `GND`. No fuse on board.
- Strip power switch BTS7002-1EPP between `VBAT` and `12V_LED`; `IN` from GPIO10 (`LED_PWR_EN`, pulled down), `IS` to GPIO2 (`LED_ISENSE`).
- Logic: AP63200 buck at 5.4 V → SS34 → `5V`; USB VBUS → SS34 → `5V`; AP7361C-33 LDO → `3V3`.
- CAN TJA1051T/3, K-line TJA1027T/20, one OBD connector JST XH 4-pin: CANH, CANL, K, GND.
- IMU LSM6DS3TR-C at I2C 0x6A. microSD push-push on SDMMC 4-bit.
- ESP32-S3-WROOM-1-N16R2. Pin map exactly as the spec table (GPIO 4,5,6,7,15,16 = LED1..6; 8/9 I2C; 10 LED_PWR_EN; 11 STATUS; 12/13 K_RX/K_TX; 14 CAN_STB; 17 IMU_INT1; 18/21 CAN_TX/CAN_RX; 19/20 USB; 35/36 UART2; 38..42,47 SD; 48 SD_CD; 43/44 UART0; 0 BOOT; 3/45/46 unused).
- Board 110×75 mm, 2 layers, 2 oz, M3 holes at (4,4) (106,4) (4,71) (106,71). Strip connectors along the top edge, XT60/USB/OBD along the bottom edge, ESP antenna hanging over the left edge.
- Every part carries `LCSC`, `MPN`, `Manufacturer` fields. JLCPCB assembly; prefer JLC basic/preferred parts.
- Output lives in `PCB/v2/`. Old EasyEDA files in `PCB/` stay untouched.
- Commit messages: lowercase, short, no corporate verbs, trailer `Claude-Session: https://claude.ai/code/session_015oqRiQH5gUcw5a6jvR5Zdk`.

```python
# PCB/v2/gen/sexp.py
"""Minimal S-expression reader/writer for KiCad files."""

class Str(str):
    """A quoted string atom."""

class Sym(str):
    """A bare atom."""

def parse(text):
    i, n = 0, len(text)
    stack = [[]]
    while i < n:
        c = text[i]
        if c.isspace():
            i += 1
        elif c == '(':
            stack.append([]); i += 1
        elif c == ')':
            node = stack.pop(); stack[-1].append(node); i += 1
        elif c == '"':
            j = i + 1; buf = []
            while text[j] != '"':
                if text[j] == '\\':
                    j += 1
                buf.append(text[j]); j += 1
            stack[-1].append(Str(''.join(buf))); i = j + 1
        else:
            j = i
            while j < n and not text[j].isspace() and text[j] not in '()':
                j += 1
            stack[-1].append(Sym(text[i:j])); i = j
    return stack[0]

def _atom(a):
    if isinstance(a, Str):
        return '"' + a.replace('\\', '\\\\').replace('"', '\\"') + '"'
    return str(a)

def dumps(node, indent=0):
    if not isinstance(node, list):
        return _atom(node)
    pad = '  ' * indent
    if all(not isinstance(x, list) for x in node):
        return pad + '(' + ' '.join(_atom(x) for x in node) + ')'
    head = [ _atom(x) for x in node if not isinstance(x, list) ]
    kids = [ x for x in node if isinstance(x, list) ]
    out = pad + '(' + ' '.join(head) + '\n'
    out += '\n'.join(dumps(k, indent + 1) for k in kids)
    return out + '\n' + pad + ')'

def find(node, tag):
    return [x for x in node if isinstance(x, list) and x and x[0] == tag]

def find1(node, tag):
    hits = find(node, tag)
    return hits[0] if hits else None
```

Note: `dumps` keeps atoms before child lists, which is how KiCad orders them; KiCad reads either.

# PCB/v2/gen/model.py
"""Design model: every part and every pin's net. Source of truth for sch + pcb."""
from dataclasses import dataclass, field

@dataclass
class Part:
    ref: str
    value: str
    symbol: str
    footprint: str
    pins: dict
    sheet: str
    lcsc: str = ""
    mpn: str = ""
    mfr: str = ""
    rot: int = 0
    dnp: bool = False

POWER_NETS = ("VBAT", "GND", "12V_LED", "5V", "3V3", "VBUS_USB")

PARTS: list = []

def _add(*parts):
    PARTS.extend(parts)

def R(ref, value, n1, n2, sheet, size="0603", lcsc="", mpn="", mfr=""):
    fp = {"0402": "Resistor_SMD:R_0402_1005Metric", "0603": "Resistor_SMD:R_0603_1608Metric",
          "0805": "Resistor_SMD:R_0805_2012Metric", "1206": "Resistor_SMD:R_1206_3216Metric",
          "2512": "Resistor_SMD:R_2512_6332Metric"}[size]
    return Part(ref, value, "Device:R", fp, {"1": n1, "2": n2}, sheet, lcsc, mpn, mfr)

def C(ref, value, n1, n2, sheet, size="0603", lcsc="", mpn="", mfr=""):
    fp = {"0402": "Capacitor_SMD:C_0402_1005Metric", "0603": "Capacitor_SMD:C_0603_1608Metric",
          "0805": "Capacitor_SMD:C_0805_2012Metric", "1210": "Capacitor_SMD:C_1210_3225Metric"}[size]
    return Part(ref, value, "Device:C", fp, {"1": n1, "2": n2}, sheet, lcsc, mpn, mfr)

def nets():
    out = {}
    for p in PARTS:
        for pin, net in p.pins.items():
            if net != "NC":
                out.setdefault(net, []).append((p.ref, pin))
    return out

def by_ref(ref):
    return next(p for p in PARTS if p.ref == ref)

def parts_on(sheet):
    return [p for p in PARTS if p.sheet == sheet]

# PCB/v2/gen/libs.py
import os, functools
from gen import env, sexp, model

@functools.lru_cache(None)
def _lib(libname):
    path = env.CUSTOM_SYM if libname == "da2" else os.path.join(env.SYMBOL_DIR, libname + ".kicad_sym")
    with open(path) as f:
        return sexp.parse(f.read())[0]

def _raw_symbol(libname, name):
    for node in sexp.find(_lib(libname), "symbol"):
        if node[1] == name:
            return node
    raise KeyError(f"{libname}:{name}")

def symbol_def(lib_id):
    """Return the symbol node renamed to lib_id, following 'extends' one level."""
    libname, name = lib_id.split(":", 1)
    node = _raw_symbol(libname, name)
    ext = sexp.find1(node, "extends")
    if ext:
        base = _raw_symbol(libname, ext[1])
        merged = [x for x in base if not (isinstance(x, list) and x[0] == "property")]
        merged += [x for x in node if isinstance(x, list) and x[0] == "property"]
        node = merged
    out = list(node)
    out[1] = sexp.Str(lib_id)
    # sub-units are named "<name>_<unit>_<style>"; rename their prefix too
    for i, x in enumerate(out):
        if isinstance(x, list) and x[0] == "symbol":
            y = list(x); y[1] = sexp.Str(lib_id.split(":")[1].join([lib_id.split(":")[0] + ":", ""]) if False else y[1].replace(name, lib_id, 1)); out[i] = y
    return out

def symbol_pins(lib_id):
    node = symbol_def(lib_id)
    pins = {}
    def walk(n):
        for x in n:
            if isinstance(x, list):
                if x[0] == "pin":
                    at = sexp.find1(x, "at"); num = sexp.find1(x, "number")[1]; nm = sexp.find1(x, "name")[1]
                    pins[num] = (float(at[1]), float(at[2]), float(at[3]) if len(at) > 3 else 0.0, nm, x[1])
                walk(x)
    walk(node)
    return pins

def footprint_path(fp_id):
    libname, name = fp_id.split(":", 1)
    d = env.CUSTOM_FP if libname == "da2" else os.path.join(env.FOOTPRINT_DIR, libname + ".pretty")
    return d, name

def all_symbols_resolve():
    bad = []
    for p in model.PARTS:
        try:
            symbol_def(p.symbol)
        except (KeyError, FileNotFoundError):
            bad.append(p.symbol)
    return bad

```python
# PCB/v2/gen/sch_writer.py
import os, uuid, math, json
from gen import env, sexp, libs, model
from gen.sexp import Str, Sym

SCH_VERSION = "20250114"     # from Step 1
GRID = 2.54
SHEETS = ["power", "mcu", "leds", "comms"]
ROOT_UUID = "0d6f1c2a-0000-4000-8000-000000000001"
SHEET_UUID = {s: f"0d6f1c2a-0000-4000-8000-00000000001{i}" for i, s in enumerate(SHEETS)}

def _u():
    return Str(str(uuid.uuid4()))

def _font(size=1.27):
    return ["effects", ["font", ["size", size, size]]]

def _prop(name, value, x, y, hide=False):
    eff = ["effects", ["font", ["size", 1.27, 1.27]]]
    if hide:
        eff.append(["hide", Sym("yes")])
    return ["property", Str(name), Str(value), ["at", x, y, 0], eff]

def _rot(px, py, rot):
    """Symbol-space (y up) -> sheet offset (y down) with rotation."""
    a = math.radians(rot)
    rx = px * math.cos(a) - py * math.sin(a)
    ry = px * math.sin(a) + py * math.cos(a)
    return rx, -ry

def _snap(v):
    return round(round(v / GRID) * GRID, 2)

def _place_part(part, x, y, sheet):
    """Return list of sexp nodes: symbol instance, wires, labels, no_connects."""
    nodes = []
    pins = libs.symbol_pins(part.symbol)
    sym = ["symbol", ["lib_id", Str(part.symbol)], ["at", x, y, part.rot], ["unit", 1],
           ["exclude_from_sim", Sym("no")], ["in_bom", Sym("no" if part.dnp else "yes")],
           ["on_board", Sym("yes")], ["dnp", Sym("yes" if part.dnp else "no")], ["uuid", _u()],
           _prop("Reference", part.ref, x, y - 2.54 * 4), _prop("Value", part.value, x, y + 2.54 * 4),
           _prop("Footprint", part.footprint, x, y, hide=True), _prop("Datasheet", "", x, y, hide=True),
           _prop("Description", "", x, y, hide=True),
           _prop("LCSC", part.lcsc, x, y, hide=True), _prop("MPN", part.mpn, x, y, hide=True),
           _prop("Manufacturer", part.mfr, x, y, hide=True)]
    for num in pins:
        sym.append(["pin", Str(num), ["uuid", _u()]])
    sym.append(["instances", ["project", Str(env.PROJECT_NAME),
                ["path", Str(f"/{ROOT_UUID}/{SHEET_UUID[sheet]}"), ["reference", Str(part.ref)], ["unit", 1]]]])
    nodes.append(sym)
    for num, (px, py, ang, name, etype) in pins.items():
        dx, dy = _rot(px, py, part.rot)
        ex, ey = _snap(x + dx), _snap(y + dy)          # pin end point on the sheet
        net = part.pins[num]
        if net == "NC":
            nodes.append(["no_connect", ["at", ex, ey], ["uuid", _u()]])
            continue
        # direction the pin points away from the body: opposite of the pin's own angle
        out_ang = (ang + part.rot) % 360        # pin angle 0 means pin drawn from end toward +x (into body)
        away = {0: (-1, 0), 180: (1, 0), 90: (0, 1), 270: (0, -1)}[int(out_ang)]
        wx, wy = _snap(ex + away[0] * GRID), _snap(ey + away[1] * GRID)
        nodes.append(["wire", ["pts", ["xy", ex, ey], ["xy", wx, wy]],
                      ["stroke", ["width", 0], ["type", Sym("default")]], ["uuid", _u()]])
        lab_ang = {(-1, 0): 180, (1, 0): 0, (0, 1): 270, (0, -1): 90}[away]
        nodes.append(["global_label", Str(net), ["shape", Sym("passive")], ["at", wx, wy, lab_ang],
                      ["fields_autoplaced", Sym("yes")],
                      ["effects", ["font", ["size", 1.27, 1.27]], ["justify", Sym("left" if lab_ang in (0, 270) else "right")]],
                      ["uuid", _u()],
                      ["property", Str("Intersheetrefs"), Str("${INTERSHEET_REFS}"), ["at", wx, wy, 0],
                       ["effects", ["font", ["size", 1.27, 1.27]], ["hide", Sym("yes")]]]])
    return nodes

def _bbox(part):
    pins = libs.symbol_pins(part.symbol).values()
    xs = [p[0] for p in pins] or [0]; ys = [p[1] for p in pins] or [0]
    return (max(xs) - min(xs)) + 15, (max(ys) - min(ys)) + 15

def _sheet_doc(sheet):
    parts = model.parts_on(sheet)
    doc = ["kicad_sch", ["version", Sym(SCH_VERSION)], ["generator", Str("da2")], ["generator_version", Str("9.0")],
           ["uuid", Str(SHEET_UUID[sheet])], ["paper", Str("A2")]]
    lib_symbols = ["lib_symbols"]
    seen = set()
    for p in parts:
        if p.symbol not in seen:
            lib_symbols.append(libs.symbol_def(p.symbol)); seen.add(p.symbol)
    if sheet == "power":
        lib_symbols.append(libs.symbol_def("power:PWR_FLAG")); seen.add("power:PWR_FLAG")
    doc.append(lib_symbols)
    x, y, row_h = 30.48, 40.64, 0
    for p in parts:
        w, h = _bbox(p)
        if x + w > 560:
            x, y, row_h = 30.48, y + row_h + 10.16, 0
        doc += _place_part(p, _snap(x + w / 2), _snap(y + h / 2), sheet)

Details to get right while running this:
- The pin end point: in KiCad symbol pins, `(at X Y ANGLE)` is the *connection end* of the pin and the angle is the direction toward the body. So the wire stub must go *away* from the body: for angle 0 (pin points right, body to the right) the stub goes left. The `away` table above encodes that; verify with one resistor by opening the sheet in Eeschema or by netlist test failing with `Net-(R1-Pad1)` style names.
- The netlist for a net that is only connected through labels named `VBAT` is `VBAT`; nets with no label become `Net-(Rx-Padn)`. `test_netlist_matches_model` fails loudly if any stub misses its pin.
- `global_label` shape `passive` avoids ERC input/output conflicts between labels.
- `PWR_FLAG` symbol pin number is `1`; the flag placement uses the same `_place_part` helper so the flag's pin gets a label.

        s = (math.cos(math.radians(a0)) * r, math.sin(math.radians(a0)) * r)
        m = (math.cos(math.radians(a0 + 45)) * r, math.sin(math.radians(a0 + 45)) * r)
        e = (math.cos(math.radians(a0 + 90)) * r, math.sin(math.radians(a0 + 90)) * r)
        arc.SetArcGeometry(MM(cx + s[0], cy + s[1]), MM(cx + m[0], cy + m[1]), MM(cx + e[0], cy + e[1])); board.Add(arc)

def _holes(board):
    for i, (x, y) in enumerate(placement.HOLES, 1):
        fp = pcbnew.FootprintLoad(os.path.join(env.FOOTPRINT_DIR, "MountingHole.pretty"), "MountingHole_3.2mm_M3")
        fp.SetReference(f"H{i}"); fp.SetValue("M3"); fp.SetPosition(MM(x, y))
        fp.SetAttributes(fp.GetAttributes() | pcbnew.FP_EXCLUDE_FROM_BOM | pcbnew.FP_EXCLUDE_FROM_POS_FILES)
        board.Add(fp)

def _nets(board):
    out = {}
    for name in model.nets():
        n = pcbnew.NETINFO_ITEM(board, name); board.Add(n); out[name] = n
    return out

def _footprints(board, nets):
    for p in model.PARTS:
        libdir, name = libs.footprint_path(p.footprint)
        fp = pcbnew.FootprintLoad(libdir, name)
        assert fp is not None, p.footprint
        fp.SetReference(p.ref); fp.SetValue(p.value)
        fp.SetField("LCSC", p.lcsc) if hasattr(fp, "SetField") else None
        x, y, rot, side = placement.PLACE[p.ref]
        fp.SetPosition(MM(x, y)); fp.SetOrientationDegrees(rot)
        if side == "B":
            fp.Flip(MM(x, y), False)
        for pad in fp.Pads():
            net = p.pins.get(pad.GetNumber(), "NC")
            if net != "NC":
                pad.SetNet(nets[net])
        board.Add(fp)

def build(path):
    board = pcbnew.BOARD()
    board.SetFileName(path)
    ds = board.GetDesignSettings()
    ds.SetCopperLayerCount(2)
    ds.SetBoardThickness(pcbnew.FromMM(1.6))
    _outline(board); _holes(board)
    nets = _nets(board); _footprints(board, nets)
    pcbnew.SaveBoard(path, board)
    write_project_netclasses()

if __name__ == "__main__":
    build(os.path.join(env.PROJECT_DIR, env.PROJECT_NAME + ".kicad_pcb"))

def _zone(board, net, layer, pts, priority=0, clearance=0.3, min_w=0.25):
    z = pcbnew.ZONE(board)
    z.SetLayer(layer); z.SetNet(net); z.SetAssignedPriority(priority)
    z.SetLocalClearance(pcbnew.FromMM(clearance)); z.SetMinThickness(pcbnew.FromMM(min_w))
    z.SetPadConnection(pcbnew.ZONE_CONNECTION_FULL)
    z.SetIsRuleArea(False)
    outline = z.Outline(); outline.NewOutline()
    for x, y in pts:
        outline.Append(pcbnew.FromMM(x), pcbnew.FromMM(y))

def _zones(board, nets):

    _zone(board, nets["GND"], pcbnew.F_Cu, full, priority=0)

    vbat = [(6, 52), (44, 52), (44, 75 - 0.5), (6, 75 - 0.5)]              # around XT60, caps, TVS
    _zone(board, nets["VBAT"], pcbnew.F_Cu, vbat, priority=1, clearance=0.4)
    vbat_b = [(6, 40), (95, 40), (95, 75 - 0.5), (6, 75 - 0.5)]           # bottom-side VBAT run to U2 tab
    _zone(board, nets["VBAT"], pcbnew.B_Cu, vbat_b, priority=1, clearance=0.4)
    tab = [(86, 19), (93.5, 19), (93.5, 32), (86, 32)]                     # U2 VS tab area, both sides
    _zone(board, nets["VBAT"], pcbnew.F_Cu, tab, priority=2, clearance=0.4)
    _zone(board, nets["VBAT"], pcbnew.B_Cu, tab, priority=2, clearance=0.4)

def _via(board, net, x, y, d=0.8, drill=0.4):
    v = pcbnew.PCB_VIA(board); v.SetPosition(MM(x, y)); v.SetWidth(pcbnew.FromMM(d)); v.SetDrill(pcbnew.FromMM(drill))
    v.SetViaType(pcbnew.VIATYPE_THROUGH); v.SetLayerPair(pcbnew.F_Cu, pcbnew.B_Cu); v.SetNet(net); board.Add(v)

def _stitch(board, nets):
    for x in (88, 89.5, 91, 92.5):          # under/next to U2 tab (tab center 94,26 ± 1.6 -> vias in the copper around it)
        for y in (21, 22.5, 24, 25.5, 27, 28.5, 30):
            _via(board, nets["VBAT"], x, y)
    for x in range(12, 104, 8):             # 12V_LED bus F<->B
        _via(board, nets["12V_LED"], x, 12.5)
    for x in range(8, 105, 10):             # GND stitching along the strip connector row and elsewhere
        _via(board, nets["GND"], x, 8.5)
        _via(board, nets["GND"], x, 38.0)

def _fill(board):
    filler = pcbnew.ZONE_FILLER(board)
    filler.Fill(board.Zones())

    subprocess.run([env.KICAD_CLI, "pcb", "export", "pos", "--format", "csv", "--units", "mm", "--side", "both",
                    "--use-drill-file-origin", "-o", pos, PCB], check=True)
    rows = list(csv.DictReader(open(pos)))
    with open(os.path.join(FAB, "CPL.csv"), "w", newline="") as f:
        w = csv.writer(f); w.writerow(["Designator", "Mid X", "Mid Y", "Layer", "Rotation"])
        for r in rows:
            if r["Ref"].startswith("H"):
                continue
            w.writerow([r["Ref"], f'{float(r["PosX"]):.3f}mm', f'{float(r["PosY"]):.3f}mm',
                        "Top" if r["Side"].lower().startswith("top") else "Bottom", r["Rot"]])

def renders():
    for side, layers in (("top", "F.Cu,F.SilkS,F.Mask,Edge_Cuts"), ("bottom", "B.Cu,B.SilkS,B.Mask,Edge_Cuts")):
        subprocess.run([env.KICAD_CLI, "pcb", "export", "svg", "--layers", layers, "--page-size-mode", "2",
                        *( ["--mirror"] if side == "bottom" else []),
                        "-o", os.path.join(FAB, f"render-{side}.svg"), PCB], check=True)
    subprocess.run([env.KICAD_CLI, "sch", "export", "pdf", "-o", os.path.join(FAB, "schematic.pdf"), SCH], check=True)

def export_all():
    os.makedirs(FAB, exist_ok=True)
    gerbers(); bom(); cpl(); renders()

if __name__ == "__main__":
    export_all()
```
